const SafeDonut = ({ data = {}, total = 0, title = "" }) => {
  const entries = useMemo(() => 
    Object.entries(data).filter(([_, v]) => typeof v === 'number' && v > 0), 
  [data]);

  let currentOffset = 0;

  if (total <= 0 || entries.length === 0) {
    return (
      <div className="bg-white p-10 rounded-[2.5rem] border-2 border-dashed border-slate-100 text-center flex flex-col items-center justify-center min-h-[220px]">
        <PieChart className="w-10 h-10 text-slate-200 mb-2" />
        <p className="text-[10px] font-black uppercase text-slate-300 tracking-widest">{title}</p>
        <p className="text-[9px] text-slate-400 italic">Esperando datos...</p>
      </div>
    );
  }

  return (
    <div className="bg-white p-6 rounded-[2.5rem] border border-slate-100 shadow-sm flex flex-col items-center animate-in zoom-in-95">
      <h4 className="text-[9px] font-black text-slate-400 uppercase tracking-widest mb-6">{title}</h4>
      <div className="relative w-28 h-28 mb-6">
        <svg viewBox="0 0 32 32" className="w-full h-full -rotate-90">
          <circle cx="16" cy="16" r="14" fill="transparent" stroke="#f8fafc" strokeWidth="4" />
          {entries.map(([cat, val], i) => {
            const percentage = (val / total) * 100;
            const dash = `${percentage} 100`;
            const offset = -currentOffset;
            currentOffset += percentage;
            return (
              <circle key={cat} cx="16" cy="16" r="14" fill="transparent" stroke={CONFIG.colors[i % CONFIG.colors.length]} strokeWidth="4.5" strokeDasharray={dash} strokeDashoffset={offset} strokeLinecap="round" />
            );
          })}
        </svg>
        <div className="absolute inset-0 flex flex-col items-center justify-center leading-none">
          <span className="text-[10px] font-black text-slate-800">${total.toFixed(0)}</span>
        </div>
      </div>
      <div className="w-full space-y-1.5 px-1 text-left">
        {entries.map(([cat, val], i) => (
          <div key={cat} className="flex justify-between items-center text-[8px] font-bold uppercase truncate">
            <div className="flex items-center gap-1.5 truncate max-w-[70%]">
              <div className="w-1.5 h-1.5 rounded-full flex-shrink-0" style={{backgroundColor: CONFIG.colors[i % CONFIG.colors.length]}}></div>
              <span className="text-slate-500 truncate">{String(cat)}</span>
            </div>
            <span className="text-slate-900 font-black">${Number(val).toFixed(2)}</span>
          </div>
        ))}
      </div>
    </div>
  );
};

            <p className="text-[9px] font-black text-emerald-600 uppercase border-l-2 border-emerald-500 pl-2">Ingresos Consolidados</p>
            {Object.entries(safeStats.byIn || {}).map(([c, v]) => v > 0 && (

// 3. APLICACIÓN PRINCIPAL (UNIFICADA)

export default function App() {
  const [user, setUser] = useState(null);
  const [activeTab, setActiveTab] = useState('inicio');
  const [isPaperMode, setIsPaperMode] = useState(false);
  const [showAdd, setShowAdd] = useState(false);
  const [showFilter, setShowFilter] = useState(false);
  const [showProfile, setShowProfile] = useState(false); 
  const [loading, setLoading] = useState(true);
  
  // Estados de Navegación Drill-Down (Blindados)
  const [activeProjectDetail, setActiveProjectDetail] = useState(null); 
  const [activeFundDetail, setActiveFundDetail] = useState(null); 
  
  const [msgAlert, setMsgAlert] = useState("");
  const [confirmDel, setConfirmDel] = useState(null);

  const [church, setChurch] = useState("");
  const [userEmail, setUserEmail] = useState("");
  const [projects, setProjects] = useState([]);
  const [txs, setTxs] = useState([]);

  const [selectedMonths, setSelectedMonths] = useState(['Todos']); 
  const [selectedYear, setSelectedYear] = useState('2026');

  const [form, setForm] = useState({ 
    desc: '', monto: '', tipo: 'in', cat: 'Ofrenda', 
    projectId: '', isProjectAssign: false, fundSource: 'Ofrenda', 
    resp: '', fecha: new Date().toISOString().split('T')[0] 
  });

  // --- NUBE: FIREBASE ---
  useEffect(() => {
    const initAuth = async () => {
      try {
        await signInAnonymously(auth);
      } catch (err) { console.error("Error Auth:", err); }
    };
    initAuth();
    return onAuthStateChanged(auth, (u) => {
      setUser(u);
      if (u) setLoading(false);
    });
  }, []);

  useEffect(() => {
    if (!user) return;
    const unsubProfile = onSnapshot(doc(db, 'artifacts', appId, 'users', user.uid, 'profile', 'main'), (s) => {
      if (s.exists()) { setChurch(s.data().church || ""); setUserEmail(s.data().email || ""); }
    });
    const unsubProj = onSnapshot(collection(db, 'artifacts', appId, 'users', user.uid, 'projects'), (s) => {
      setProjects(s.docs.map(d => ({ id: d.id, ...d.data() })));
    });
    const unsubTxs = onSnapshot(collection(db, 'artifacts', appId, 'users', user.uid, 'transactions'), (s) => {
      const list = s.docs.map(d => ({ id: d.id, ...d.data() }));
      setTxs(list.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0)));
    });
    return () => { unsubProfile(); unsubProj(); unsubTxs(); };
  }, [user]);

  // --- MOTOR DE CÁLCULO ---
  const calculated = useMemo(() => {
    const s = { in: 0, out: 0, bal: 0, byIn: {}, byOut: {} };
    const fundBalances = {};
    const projSums = {};
    
    (projects || []).forEach(p => { if(p.id) projSums[p.id] = 0; });
    CONFIG.categories.in.forEach(c => fundBalances[c] = 0);
    CONFIG.categories.out.forEach(c => s.byOut[c] = 0);

    txs.forEach(t => {
      const val = parseFloat(t.monto) || 0;
      const d = new Date(t.fecha);
      if (isNaN(d.getTime())) return;
      const mName = CONFIG.months[d.getMonth()];
      const yName = d.getFullYear().toString();
      const isVisible = (selectedMonths.includes('Todos') || selectedMonths.includes(mName)) && (selectedYear === yName);

      if (isVisible) {
        if (t.tipo === 'in') {
          s.in += val; s.bal += val;
          s.byIn[t.cat] = (s.byIn[t.cat] || 0) + val;
          fundBalances[t.cat] = (fundBalances[t.cat] || 0) + val;
          if (t.projectId && projSums[t.projectId] !== undefined) projSums[t.projectId] += val;
        } else {
          s.out += val; s.bal -= val;
          s.byOut[t.cat] = (s.byOut[t.cat] || 0) + val;
          const src = t.fundSource || 'Ofrenda';
          fundBalances[src] = (fundBalances[src] || 0) - val;
          if (t.projectId && projSums[t.projectId] !== undefined) projSums[t.projectId] -= val;
        }
      }
    });
    return { stats: s, fundBalances, projectContributions: projSums };
  }, [txs, projects, selectedMonths, selectedYear]);

  // Acciones
  const handleSave = async (e) => {
    e.preventDefault();
    if (!user) return;
    const finalData = { 
      ...form, 
      monto: parseFloat(form.monto) || 0, 
      projectId: form.isProjectAssign ? form.projectId : '', 
      createdAt: Date.now() 
    };
    await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'transactions'), finalData);
    setShowAdd(false);
    setMsgAlert("✅ Sincronizado");
    setForm({ ...form, desc: '', monto: '', isProjectAssign: false });
    setTimeout(() => setMsgAlert(""), 3000);
  };

  const syncProfile = async () => {
    if (!user) return;
    await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'profile', 'main'), { church, email: userEmail });
    setMsgAlert("✅ Datos Guardados");
    setShowProfile(false);
    setTimeout(() => setMsgAlert(""), 3000);
  };

  const handleWhatsApp = () => {
    const text = `📊 *ESTADO FINANCIERO: ${church.toUpperCase()}*\n💰 SALDO: $${calculated.stats.bal.toFixed(2)}\n📉 GASTOS: $${calculated.stats.out.toFixed(2)}\n━━━━━━━━━━━━━━━━━━\n✅ _Gestor Pro v76.0_`;
    window.open(`https://wa.me/?text=${encodeURIComponent(text)}`, '_blank');
  };

  const safeSpending = Math.max((calculated.stats.bal - Object.values(calculated.projectContributions).reduce((a,b)=>(a||0)+(b||0),0)) * 0.85, 0);

  if (loading) return (
    <div className="min-h-screen bg-slate-900 flex flex-col items-center justify-center text-white p-6">
      <Loader2 className="w-12 h-12 animate-spin text-indigo-500 mb-4" />
      <p className="text-[10px] font-black uppercase tracking-widest animate-pulse text-center">Iniciando sistema ministerial...</p>
    </div>
  );

  if (isPaperMode) {
    return (
      <div className="min-h-screen bg-slate-100 overflow-y-auto">
        <div className="no-print bg-slate-900 p-6 flex justify-between items-center text-white sticky top-0 z-[1000]">
          <span className="text-[10px] font-black uppercase">Reporte PDF</span>
          <button onClick={() => setIsPaperMode(false)} className="bg-indigo-600 px-6 py-2 rounded-full font-black text-[10px] uppercase">Regresar</button>
        </div>
        <div className="max-w-4xl mx-auto py-10 px-4">
          <OfficialReport church={church} stats={calculated.stats} fundBalances={calculated.fundBalances} projects={projects} projectContributions={calculated.projectContributions} />
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-[#F8FAFC] font-sans pb-36 text-slate-900 overflow-x-hidden selection:bg-indigo-100 text-left leading-tight">
      
      {/* MODAL CONFIGURACIÓN PERFIL */}
      {showProfile && (
        <div className="fixed inset-0 bg-slate-900/95 backdrop-blur-md z-[4000] flex items-center justify-center p-6 animate-in fade-in">
           <div className="bg-white w-full max-w-sm rounded-[3rem] p-10 shadow-2xl space-y-8 relative text-left leading-none">
              <button onClick={() => setShowProfile(false)} className="absolute top-8 right-8 text-slate-300 active:scale-75 leading-none text-center"><X size={24} /></button>
              <div className="flex flex-col items-center text-center space-y-2">
                 <div className="w-16 h-16 bg-indigo-50 text-indigo-600 rounded-2xl flex items-center justify-center shadow-inner"><Settings size={32} /></div>
                 <h3 className="text-lg font-black uppercase leading-tight text-slate-900">Perfil del Ministerio</h3>
                 <p className="text-[9px] text-slate-400 uppercase tracking-widest text-left leading-none">Respaldo Cloud por correo</p>
              </div>
              <div className="space-y-4 text-left leading-none">
                 <div className="space-y-1"><label className="text-[8px] font-black text-slate-400 uppercase ml-2 text-left leading-none">Nombre Iglesia</label><input type="text" className="w-full bg-slate-50 p-5 rounded-2xl border-none outline-none font-bold text-xs shadow-inner" value={church} onChange={e=>setChurch(e.target.value)} /></div>
                 <div className="space-y-1"><label className="text-[8px] font-black text-slate-400 uppercase ml-2 text-left leading-none">Correo Electrónico</label><input type="email" className="w-full bg-slate-50 p-5 rounded-2xl border-none outline-none font-bold text-xs shadow-inner" placeholder="ejemplo@iglesia.com" value={userEmail} onChange={e=>setUserEmail(e.target.value)} /></div>
                 <button onClick={syncProfile} className="w-full py-5 bg-indigo-600 text-white rounded-2xl font-black text-[10px] uppercase shadow-lg shadow-indigo-100 active:scale-95 transition-all text-center leading-none">Sincronizar</button>
              </div>
           </div>
        </div>
      )}

      {/* MODAL ELIMINAR */}
      {confirmDel && (
        <div className="fixed inset-0 bg-slate-900/90 backdrop-blur-md z-[5000] flex items-center justify-center p-6 animate-in fade-in">
           <div className="bg-white w-full max-w-xs rounded-[3rem] p-10 shadow-2xl text-center space-y-6">
              <div className="w-20 h-20 bg-rose-50 text-rose-500 rounded-full flex items-center justify-center mx-auto shadow-inner leading-none text-center"><Trash2 size={32} /></div>
              <h4 className="text-sm font-black uppercase leading-tight text-slate-900">¿Eliminar este registro?</h4>
              <div className="flex gap-3 text-center">
                 <button onClick={()=>setConfirmDel(null)} className="flex-1 py-4 rounded-2xl bg-slate-100 text-slate-500 font-black text-[10px] uppercase text-center leading-none">No</button>
                 <button onClick={async () => {
                    const col = confirmDel.type === 'tx' ? 'transactions' : 'projects';
                    await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, col, confirmDel.id));
                    setConfirmDel(null); setMsgAlert("🗑️ Borrado de Nube");
                 }} className="flex-1 py-4 rounded-2xl bg-rose-500 text-white font-black text-[10px] uppercase shadow-lg text-center leading-none">Sí, borrar</button>
              </div>
           </div>
        </div>
      )}

      {msgAlert && (
        <div className="fixed top-6 left-1/2 -translate-x-1/2 z-[1000] animate-in slide-in-from-top duration-300">
           <div className="bg-slate-900 text-white px-8 py-4 rounded-full shadow-2xl font-black text-[10px] uppercase border border-indigo-500/30 flex items-center gap-3">
              <CheckCircle2 size={16} className="text-emerald-400" /> {msgAlert}
           </div>
        </div>
      )}

      {/* HEADER */}
      <header className="bg-slate-900 text-white pt-16 pb-28 px-8 rounded-b-[4.5rem] shadow-2xl relative text-center no-print overflow-hidden border-b-8 border-indigo-600">
        <div className="absolute top-0 right-0 w-80 h-80 bg-indigo-500/10 rounded-full -mr-32 -mt-32 blur-3xl"></div>
        <button onClick={() => setShowProfile(true)} className="absolute top-10 right-10 w-12 h-12 bg-white/10 rounded-2xl flex items-center justify-center text-white border border-white/10 active:scale-75 transition-all shadow-lg leading-none text-center"><Settings size={20} /></button>
        <div className="relative z-10 text-center leading-none">
          <p className="text-[10px] font-black opacity-30 uppercase tracking-[0.4em] mb-3 italic text-center leading-none">{church || "Tesoreria"}</p>
          <h2 className="text-6xl font-black tracking-tighter leading-none mb-10 text-white text-center leading-none">${calculated.stats.bal.toLocaleString(undefined, { minimumFractionDigits: 2 })}</h2>
          <div className="flex justify-center gap-3 text-center leading-none">
             <button onClick={() => setShowFilter(true)} className="bg-white/10 px-8 py-3.5 rounded-full text-[10px] font-black uppercase tracking-widest active:scale-95 transition-all text-white border border-white/10 flex items-center gap-2 shadow-lg leading-none text-center"><Calendar size={14} /> Filtro</button>
             <button onClick={() => setMsgAlert("Caja Conciliada ✓")} className="bg-emerald-600 px-8 py-3.5 rounded-full text-[10px] font-black uppercase tracking-widest active:scale-95 transition-all text-white shadow-lg border border-white/20 text-center leading-none">Conciliar</button>
          </div>
        </div>
      </header>

      <main className="max-w-xl mx-auto p-6 -mt-16 relative z-20 text-left space-y-8">
        
        {/* VISTA DETALLE DRILL-DOWN */}
        {(activeProjectDetail || activeFundDetail) && (
           <div className="space-y-6 animate-in slide-in-from-right duration-500 pb-10 text-left leading-none">
              <div className="flex items-center gap-4 mb-4 text-left">
                 <button onClick={() => {setActiveProjectDetail(null); setActiveFundDetail(null);}} className="w-12 h-12 bg-white rounded-2xl shadow-sm border border-slate-100 flex items-center justify-center text-slate-400 active:scale-90 shadow-indigo-100/50 leading-none text-center"><ArrowLeft size={20} /></button>
                 <div className="text-left leading-tight"><h3 className="text-xl font-black text-slate-800 uppercase leading-none truncate max-w-[200px] text-left">{activeProjectDetail ? projects.find(p=>p.id===activeProjectDetail)?.name : activeFundDetail}</h3><p className="text-[10px] font-bold text-indigo-500 uppercase tracking-widest mt-1 italic leading-none text-left">Libro Auxiliar Cronológico</p></div>
              </div>
              <div className="bg-white p-8 rounded-[3rem] border border-slate-100 shadow-sm space-y-2 relative overflow-hidden text-left leading-none"><p className="text-[9px] font-black text-slate-400 uppercase tracking-widest leading-none text-left">Saldo Neto Acumulado</p><p className="text-4xl font-black text-indigo-600 tracking-tighter text-left">${activeProjectDetail ? (calculated.projectContributions[activeProjectDetail] || 0).toFixed(2) : (calculated.fundBalances[activeFundDetail] || 0).toFixed(2)}</p></div>
              <div className="space-y-4 text-left leading-none text-left">
                 <div className="bg-white rounded-[3rem] border border-slate-100 shadow-sm overflow-hidden divide-y divide-slate-50 text-left leading-none">
                    {txs.filter(t => activeProjectDetail ? t.projectId === activeProjectDetail : (t.cat === activeFundDetail || t.fundSource === activeFundDetail)).length === 0 ? (
                       <p className="p-10 text-center text-[10px] text-slate-300 uppercase italic leading-loose text-center leading-none">Sin movimientos vinculados.</p>

'bg-indigo-600 text-white border-indigo-600 shadow-lg' : 'bg-white text-indigo-600 border-indigo-100'} text-center leading-none text-center text-center text-center text-center`}>
                     {form.isProjectAssign ? "Meta Asignada ✓" : "¿Asignar a un Proyecto?"}
                   </button>
                   {form.isProjectAssign && projects.length > 0 && (
                      <div className="animate-in slide-in-from-top px-2 text-left leading-none text-left leading-none text-left text-left text-left text-left text-left text-left"><select className="w-full bg-slate-900 p-5 rounded-[2.5rem] border-none font-bold text-xs shadow-xl outline-none text-white uppercase appearance-none text-center shadow-indigo-100 text-center leading-none text-center text-center text-center text-center text-center" value={form.projectId} onChange={e => setForm({...form, projectId: e.target.value})} required>{projects.map(p => <option key={p.id} value={p.id}>{String(p.name)}</option>)}</select></div>
                   )}
                </div>
                <input type="date" className="w-full bg-slate-50 p-6 rounded-[2.5rem] border-none font-bold text-xs shadow-inner outline-none text-slate-600 text-left leading-none text-left text-left text-left text-left text-left text-left text-left" value={form.fecha} onChange={e => setForm({...form, fecha: e.target.value})} required />
              </div>
              <button type="submit" className={`w-full py-7 rounded-[3rem] text-white font-black uppercase text-xs tracking-widest mt-6 shadow-2xl transition-all active:scale-95 ${form.tipo === 'in' ? 'bg-emerald-500 shadow-emerald-500/30' : 'bg-rose-500 shadow-rose-500/30'} text-center leading-none text-center text-center text-center text-center text-center text-center`}>CONFIRMAR REGISTRO</button>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
